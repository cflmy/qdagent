# main

**task = > sys.env_get name="QDAGENT_TASK"**
**failure = > sys.env_get name="QDAGENT_FAILURE"**
**decision = > sys.env_get name="QDAGENT_DECISION"**
**constraint = > sys.env_get name="QDAGENT_CONSTRAINT"**
1. not `task`
  **task = "qdagent engineering knowledge refresh"**
1. not `failure`
  **failure = ""**
1. not `decision`
  **decision = ""**
1. not `constraint`
  **constraint = ""**

**out = > qd.learn task=`task` failure=`failure` decision=`decision` constraint=`constraint`**
**s = > json.stringify value=`out`**
> print text=`s`
*s*
