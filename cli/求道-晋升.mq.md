# main

**action = > sys.env_get name="QDAGENT_PROMOTE_ACTION"**
1. not `action`
  **action = "list"**

**slug = > sys.env_get name="QDAGENT_CANDIDATE"**
**kind = > sys.env_get name="QDAGENT_KIND"**
1. not `kind`
  **kind = "decision"**
**reason = > sys.env_get name="QDAGENT_REASON"**
1. not `reason`
  **reason = "rejected"**

1. `action` == "promote"
  1. not `slug`
    > print text="usage: qdagent promote <candidate-slug>"
    *""*
  **out = > promote.promote slug=`slug` kind=`kind`**
2. `action` == "reject"
  1. not `slug`
    > print text="usage: qdagent reject <candidate-slug>"
    *""*
  **out = > promote.reject slug=`slug` reason=`reason`**
3. *
  **status = > sys.env_get name="QDAGENT_STATUS"**
  1. not `status`
    **status = ""**
  **out = > promote.list status=`status`**

**s = > json.stringify value=`out`**
> print text=`s`
*""*
