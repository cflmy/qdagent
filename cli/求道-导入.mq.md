# main

**path = > sys.env_get name="QDAGENT_IMPORT_PATH"**
1. not `path`
  > print text=usage: QDAGENT_IMPORT_PATH=/path/to/file.txt [QDAGENT_KB_ROOT=…] [QDAGENT_IMPORT_PUBLISH=1]
  > sys.exit code=2
**kb = > sys.env_get name="QDAGENT_KB_ROOT"**
1. not `kb`
  **kb = ""**
**pub = > sys.env_get name="QDAGENT_IMPORT_PUBLISH"**
**do_pub = False**
1. `pub` == "1"
  **do_pub = True**
1. `pub` == "true"
  **do_pub = True**
**out = > imp.import_txt path=`path` kb_root=`kb` publish=`do_pub` product_gate_ok=True**
**s = > json.stringify value=`out`**
> print text=`s`
**ok = access(out, "ok")**
1. not `ok`
  > sys.exit code=1
*"ok"*
